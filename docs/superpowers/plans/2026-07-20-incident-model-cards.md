# Incident Model + Cards Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an `Incident` decision object that clusters mentions into citizen-service problems with severity, location, OPD assignment, a grounded draft narrative, and an auditable lifecycle — surfaced as a card board and detail page.

**Architecture:** New models (`OPD`, `Incident`, `IncidentEvent`) in `apps/analytics/models.py`. Detection and composition live in a new `apps/analytics/incidents.py` that mirrors the existing `apps/analytics/briefs.py` structure: threshold-based shortlisting, deterministic grounded composition behind a swappable seam, and a mandatory draft banner. Views follow the existing `war_room` / `regional` pattern, gated by a new `insiden` page in the permission matrix.

**Tech Stack:** Django (server-rendered templates, no SPA), PostgreSQL/SQLite, Django `TestCase`. No new third-party dependencies.

## Global Constraints

- **No hosted LLM.** Composition is deterministic and grounded. A hosted LLM may later replace `compose_incident_narrative()` behind the same signature, but is out of scope here.
- **Draft banner mandatory.** All four composed narrative fields must contain the banner text `**[DRAF — DIHASILKAN AI, MENUNGGU TINJAUAN MANUSIA]**`. Reuse `briefs.DRAFT_BANNER` — do not redefine the string.
- **Tenant isolation (FR-1.2).** Every incident query in a view goes through `apps.core.selectors.scope_to_tenant`. Never filter by `project` alone from a request parameter.
- **Recommendations are advisory.** The system proposes; a human assigns and changes status. No automated action.
- **Language:** All user-facing strings are Bahasa Indonesia. Code identifiers, comments, and docstrings are English, matching the existing codebase.
- **Existing models are untouched.** Do not modify `PoliticalIssue` or the taxonomy `Issue`.
- **Do not modify `apps/analytics/briefs.py`** except where a task explicitly says so (no task does).
- Tests run with: `python manage.py test apps.analytics`

```python
class OPD(models.Model):
    """An organizational unit (Dinas) an incident can be assigned to.

    Scoped per project so each tenant defines its own structure. Introduced with
    the Incident board; also the grouping key for the later per-OPD KPI view.
    """

    project = models.ForeignKey(Project, on_delete=models.CASCADE, related_name="opds")
    name = models.CharField(max_length=200)
    slug = models.SlugField()
    abbr = models.CharField(max_length=20, blank=True)
    is_active = models.BooleanField(default=True)

    class Meta:
        unique_together = ("project", "slug")
        ordering = ["name"]
        verbose_name = "OPD"
        verbose_name_plural = "OPD"

    def __str__(self):
        return self.abbr or self.name


class Incident(models.Model):
    """A citizen-service problem surfaced by social conversation, tracked as a
    decision object for leadership.

    Distinct from PoliticalIssue (a political-narrative war-room event, gated by
    neutrality controls) and from the taxonomy Issue (a classification leaf applied
    to individual mentions). An Incident *aggregates* mentions — clustered on
    taxonomy Issue x Regency — into something a Kepala Dinas can act on.

    Narrative fields are composed deterministically and always carry the draft
    banner: nothing here is published to leadership without human review.
    """

    SEVERITY_CRITICAL = "critical"
    SEVERITY_HIGH = "high"
    SEVERITY_MEDIUM = "medium"
    SEVERITY_LOW = "low"
    SEVERITY_CHOICES = [
        (SEVERITY_CRITICAL, "Kritis"),
        (SEVERITY_HIGH, "Tinggi"),
        (SEVERITY_MEDIUM, "Sedang"),
        (SEVERITY_LOW, "Rendah"),
    ]

    STATUS_OPEN = "open"
    STATUS_ASSIGNED = "assigned"
    STATUS_IN_PROGRESS = "in_progress"
    STATUS_RESOLVED = "resolved"
    STATUS_CLOSED = "closed"
    STATUS_CHOICES = [
        (STATUS_OPEN, "Terbuka"),
        (STATUS_ASSIGNED, "Ditugaskan"),
        (STATUS_IN_PROGRESS, "Ditangani"),
        (STATUS_RESOLVED, "Selesai"),
        (STATUS_CLOSED, "Ditutup"),
    ]

    project = models.ForeignKey(Project, on_delete=models.CASCADE, related_name="incidents")
    title = models.CharField(max_length=250)
    issue = models.ForeignKey(
        Issue, on_delete=models.SET_NULL, null=True, blank=True, related_name="incidents"
    )
    regency = models.ForeignKey(
        Regency, on_delete=models.SET_NULL, null=True, blank=True, related_name="incidents"
    )
    severity = models.CharField(max_length=20, choices=SEVERITY_CHOICES, default=SEVERITY_LOW)
    status = models.CharField(max_length=20, choices=STATUS_CHOICES, default=STATUS_OPEN)
    assigned_opd = models.ForeignKey(
        OPD, on_delete=models.SET_NULL, null=True, blank=True, related_name="incidents"
    )
    mentions = models.ManyToManyField(Mention, blank=True, related_name="incidents")
    mention_count = models.PositiveIntegerField(default=0)
    negative_pct = models.PositiveIntegerField(default=0)
    ai_summary = models.TextField(blank=True)
    root_cause = models.TextField(blank=True)
    impact = models.TextField(blank=True)
    recommendation = models.TextField(blank=True)
    is_confirmed = models.BooleanField(default=False)
    first_seen_at = models.DateTimeField(null=True, blank=True)
    closed_at = models.DateTimeField(null=True, blank=True)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ["-severity", "-mention_count"]
        indexes = [
            models.Index(fields=["project", "status"]),
            models.Index(fields=["severity"]),
        ]

    @property
    def is_open(self):
        return self.status not in (self.STATUS_RESOLVED, self.STATUS_CLOSED)

    def __str__(self):
        return self.title


class IncidentEvent(models.Model):
    """One entry on an incident's timeline (detection, assignment, status change,
    logged action). Together these form the auditable record of how the institution
    responded — the same role ResponseAction plays for a political narrative event.
    """

    KIND_DETECTED = "detected"
    KIND_VOLUME_UPDATE = "volume_update"
    KIND_ASSIGNED = "assigned"
    KIND_STATUS_CHANGE = "status_change"
    KIND_ACTION_LOGGED = "action_logged"
    KIND_NOTE = "note"
    KIND_CHOICES = [
        (KIND_DETECTED, "Terdeteksi"),
        (KIND_VOLUME_UPDATE, "Perubahan volume"),
        (KIND_ASSIGNED, "Penugasan"),
        (KIND_STATUS_CHANGE, "Perubahan status"),
        (KIND_ACTION_LOGGED, "Tindakan dicatat"),
        (KIND_NOTE, "Catatan"),
    ]

    incident = models.ForeignKey(Incident, on_delete=models.CASCADE, related_name="events")
    kind = models.CharField(max_length=20, choices=KIND_CHOICES)
    description = models.CharField(max_length=500)
    logged_by = models.CharField(max_length=150, blank=True)
    occurred_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ["occurred_at", "id"]

    def __str__(self):
        return f"{self.get_kind_display()} — {self.description[:40]}"
```

```python
"""Incident detection and composition.

An Incident is a citizen-service problem (jalan rusak, sampah, banjir) aggregated
from conversation so leadership can answer three questions: what is happening, how
serious is it, and what should be done.

Two properties are inherited deliberately from the Daily Brief engine (briefs.py):

* Narrative is **deterministic and grounded** — it states measured numbers and
  quotes source snippets verbatim rather than inventing prose. A hosted LLM can
  replace ``compose_incident_narrative`` behind the same signature later.
* Output is always **draft-labeled**, pending human review (FR-31.4).

Recommendations are advisory. The composer proposes an OPD; a human assigns it.
"""
from apps.governance import regulatory

from .briefs import DRAFT_BANNER
from .models import Incident

# A cluster is only worth raising to leadership if it carries real signal. Mirrors
# the thresholds the brief engine uses for the same judgement.
MIN_MENTIONS = 5
MIN_NEGATIVE_PCT = 40

# How many source snippets to quote in the summary.
MAX_SNIPPETS = 3

_RISK_TO_SEVERITY = {
    regulatory.RISK_HIGH: Incident.SEVERITY_HIGH,
    regulatory.RISK_MEDIUM: Incident.SEVERITY_MEDIUM,
    regulatory.RISK_LOW: Incident.SEVERITY_LOW,
}


def severity_for(negative_pct, volume_change_pct=0, coordinated=False):
    """Map measured signal onto an incident severity.

    Delegates to the existing regulatory risk classifier so incidents and briefs
    triage consistently, then adds the ``critical`` tier the classifier has no
    concept of: high risk *plus* a volume surge is the shape of a crisis rather
    than a steady grievance.
    """
    risk = regulatory.classify_risk(
        negative_ratio=negative_pct / 100,
        volume_change_pct=volume_change_pct,
        coordinated=coordinated,
    )
    if risk == regulatory.RISK_HIGH and volume_change_pct >= 200:
        return Incident.SEVERITY_CRITICAL
    return _RISK_TO_SEVERITY[risk]


def _location(candidate):
    return candidate.get("regency_name") or "wilayah yang belum teridentifikasi"


def compose_incident_narrative(candidate):
    """Compose the four narrative fields for one incident candidate.

    This is the seam a hosted LLM would replace. Its contract: accept a candidate
    dict, return the four fields, each carrying the draft banner.
    """
    total = candidate["total"]
    negative_pct = candidate["negative_pct"]
    location = _location(candidate)
    snippets = candidate.get("snippets") or []

    summary_lines = [
        DRAFT_BANNER,
        "",
        f"Terdapat **{total}** sebutan terkait \"{candidate['title']}\" "
        f"di **{location}**, dengan **{negative_pct}%** bersentimen negatif.",
    ]
    if snippets:
        summary_lines.append("")
        summary_lines.append("Kutipan percakapan dengan keterlibatan tertinggi:")
        for snippet in snippets[:MAX_SNIPPETS]:
            summary_lines.append(f"- \"{snippet}\"")

    if snippets:
        root_cause = (
            f"{DRAFT_BANNER}\n\n"
            "Penyebab belum diverifikasi. Indikasi awal berdasarkan kutipan warga "
            "di atas perlu dikonfirmasi melalui pemeriksaan lapangan oleh OPD terkait."
        )
    else:
        root_cause = (
            f"{DRAFT_BANNER}\n\n"
            "Penyebab belum dapat disimpulkan: tidak tersedia kutipan percakapan "
            "yang memadai. Jangan mengarang penyebab; lakukan verifikasi lapangan."
        )

    impact = (
        f"{DRAFT_BANNER}\n\n"
        f"Terpapar pada **{total}** percakapan publik di {location}. "
        f"Proporsi negatif **{negative_pct}%** menunjukkan tingkat ketidakpuasan "
        "warga yang perlu direspons."
    )

    recommendation = (
        f"{DRAFT_BANNER}\n\n"
        f"Disarankan menugaskan OPD yang membidangi \"{candidate['title']}\" untuk "
        f"verifikasi lapangan di {location}, lalu memperbarui status insiden ini "
        "setelah penanganan dimulai. Penugasan dilakukan oleh pejabat berwenang, "
        "bukan secara otomatis oleh sistem."
    )

    return {
        "ai_summary": "\n".join(summary_lines),
        "root_cause": root_cause,
        "impact": impact,
        "recommendation": recommendation,
    }
```

```python
DEFAULT_WINDOW_DAYS = 7


def shortlist_candidates(project, since=None, window_days=DEFAULT_WINDOW_DAYS):
    """Group recent labeled conversation into candidate incidents.

    Clusters on (taxonomy Issue x Regency): the same complaint in two different
    kecamatan is two incidents, because they are assigned and resolved separately.
    """
    if since is None:
        since = timezone.now() - timedelta(days=window_days)

    rows = (
        IssueLabel.objects.filter(
            mention__project=project, mention__posted_at__gte=since
        )
        .values("issue__id", "issue__name", "mention__regency__id", "mention__regency__name")
        .annotate(
            total=Count("mention__id"),
            negative=Count(
                "mention__id", filter=Q(mention__sentiment=Mention.SENTIMENT_NEGATIVE)
            ),
        )
        .order_by("-negative", "-total")
    )

    candidates = []
    for row in rows:
        total = row["total"]
        negative_pct = round((row["negative"] / total) * 100) if total else 0
        if total < MIN_MENTIONS or negative_pct < MIN_NEGATIVE_PCT:
            continue

        mentions = list(
            Mention.objects.filter(
                project=project,
                posted_at__gte=since,
                issue_label__issue_id=row["issue__id"],
                regency_id=row["mention__regency__id"],
            )
            .order_by("-engagement_count")
            .values_list("id", "content_snippet")
        )
        candidates.append({
            "title": row["issue__name"],
            "issue_id": row["issue__id"],
            "regency_id": row["mention__regency__id"],
            "regency_name": row["mention__regency__name"] or "",
            "total": total,
            "negative_pct": negative_pct,
            "snippets": [snippet for _id, snippet in mentions[:MAX_SNIPPETS]],
            "mention_ids": [mention_id for mention_id, _snippet in mentions],
        })
    return candidates


@transaction.atomic
def refresh_candidates(project, since=None, window_days=DEFAULT_WINDOW_DAYS):
    """Upsert candidate incidents for a project, returning the touched incidents.

    A confirmed incident is left alone entirely: once an operator has promoted and
    edited it, re-running detection must not clobber their judgement with a fresh
    machine composition.
    """
    touched = []
    for candidate in shortlist_candidates(project, since=since, window_days=window_days):
        incident = Incident.objects.filter(
            project=project,
            issue_id=candidate["issue_id"],
            regency_id=candidate["regency_id"],
        ).first()

        if incident is not None and incident.is_confirmed:
            continue

        narrative = compose_incident_narrative(candidate)
        severity = severity_for(negative_pct=candidate["negative_pct"])

        if incident is None:
            incident = Incident.objects.create(
                project=project,
                title=candidate["title"],
                issue_id=candidate["issue_id"],
                regency_id=candidate["regency_id"],
                severity=severity,
                mention_count=candidate["total"],
                negative_pct=candidate["negative_pct"],
                first_seen_at=timezone.now(),
                **narrative,
            )
            IncidentEvent.objects.create(
                incident=incident,
                kind=IncidentEvent.KIND_DETECTED,
                description=(
                    f"Terdeteksi otomatis: {candidate['total']} sebutan, "
                    f"{candidate['negative_pct']}% negatif."
                ),
                logged_by="sistem",
            )
        else:
            previous = incident.mention_count
            incident.title = candidate["title"]
            incident.severity = severity
            incident.mention_count = candidate["total"]
            incident.negative_pct = candidate["negative_pct"]
            for field, value in narrative.items():
                setattr(incident, field, value)
            incident.save()
            if candidate["total"] != previous:
                IncidentEvent.objects.create(
                    incident=incident,
                    kind=IncidentEvent.KIND_VOLUME_UPDATE,
                    description=f"Volume berubah dari {previous} menjadi {candidate['total']} sebutan.",
                    logged_by="sistem",
                )

        incident.mentions.set(candidate["mention_ids"])
        touched.append(incident)
    return touched
```

```python
class InvalidTransition(Exception):
    """The requested status change is not permitted from the current status."""


# An incident may not skip stages: a resolution that was never assigned or worked
# leaves a KPI record that does not describe what actually happened. Closed is
# terminal — a recurrence is a new incident, not a reopened one.
ALLOWED_TRANSITIONS = {
    Incident.STATUS_OPEN: {Incident.STATUS_ASSIGNED, Incident.STATUS_CLOSED},
    Incident.STATUS_ASSIGNED: {Incident.STATUS_IN_PROGRESS, Incident.STATUS_CLOSED},
    Incident.STATUS_IN_PROGRESS: {Incident.STATUS_RESOLVED, Incident.STATUS_CLOSED},
    Incident.STATUS_RESOLVED: {Incident.STATUS_CLOSED},
    Incident.STATUS_CLOSED: set(),
}


def promote(incident, by):
    """Confirm a detected candidate as a real incident (FR: human gate)."""
    incident.is_confirmed = True
    incident.save(update_fields=["is_confirmed", "updated_at"])
    IncidentEvent.objects.create(
        incident=incident,
        kind=IncidentEvent.KIND_ACTION_LOGGED,
        description="Kandidat dikonfirmasi sebagai insiden.",
        logged_by=by,
    )
    return incident


def assign(incident, opd, by):
    """Assign an incident to an OPD, advancing it out of `open`."""
    incident.assigned_opd = opd
    fields = ["assigned_opd", "updated_at"]
    if incident.status == Incident.STATUS_OPEN:
        incident.status = Incident.STATUS_ASSIGNED
        fields.append("status")
    incident.save(update_fields=fields)
    IncidentEvent.objects.create(
        incident=incident,
        kind=IncidentEvent.KIND_ASSIGNED,
        description=f"Ditugaskan ke {opd}.",
        logged_by=by,
    )
    return incident


def set_status(incident, status, by):
    """Move an incident to ``status``, refusing transitions that skip the lifecycle."""
    if status not in ALLOWED_TRANSITIONS.get(incident.status, set()):
        raise InvalidTransition(
            f"Tidak dapat mengubah status dari "
            f"{incident.get_status_display()} ke {dict(Incident.STATUS_CHOICES).get(status, status)}."
        )

    previous_display = incident.get_status_display()
    incident.status = status
    fields = ["status", "updated_at"]
    if status == Incident.STATUS_CLOSED:
        incident.closed_at = timezone.now()
        fields.append("closed_at")
    incident.save(update_fields=fields)

    IncidentEvent.objects.create(
        incident=incident,
        kind=IncidentEvent.KIND_STATUS_CHANGE,
        description=f"Status berubah dari {previous_display} menjadi {incident.get_status_display()}.",
        logged_by=by,
    )
    return incident


def log_note(incident, description, by):
    """Append a free-text note to the timeline without changing status."""
    return IncidentEvent.objects.create(
        incident=incident,
        kind=IncidentEvent.KIND_NOTE,
        description=description,
        logged_by=by,
    )
```

```python
def incident_board(request, window="7", opd_slug="", severity=""):
    """Incidents for the board, split into operator candidates and confirmed items.

    Both halves come from the same tenant-scoped queryset: candidates are shown
    alongside confirmed incidents rather than hidden in a separate queue, so the
    operator's review work stays visible in context.
    """
    from datetime import timedelta

    from django.utils import timezone

    from apps.core.selectors import scope_to_tenant

    from .models import Incident, OPD

    queryset = scope_to_tenant(
        Incident.objects.select_related("assigned_opd", "regency"), request
    )
    # Same tenant gate as the incidents themselves, so the OPD filter can never
    # offer another tenant's units.
    opds = scope_to_tenant(OPD.objects.filter(is_active=True), request)

    try:
        days = int(window)
    except (TypeError, ValueError):
        days = 7
    if days > 0:
        queryset = queryset.filter(created_at__gte=timezone.now() - timedelta(days=days))

    if opd_slug:
        queryset = queryset.filter(assigned_opd__slug=opd_slug)
    if severity:
        queryset = queryset.filter(severity=severity)

    confirmed = list(queryset.filter(is_confirmed=True))
    candidates = list(queryset.filter(is_confirmed=False))

    return {
        "confirmed": confirmed,
        "candidates": candidates,
        "opds": opds,
        "counts": {
            "confirmed": len(confirmed),
            "candidates": len(candidates),
            "critical": sum(1 for i in confirmed if i.severity == Incident.SEVERITY_CRITICAL),
            "open": sum(1 for i in confirmed if i.is_open),
        },
    }


def incident_detail(request, incident_id):
    """One incident with its timeline and related mentions, tenant-scoped."""
    from apps.core.selectors import scope_to_tenant

    from .models import Incident

    return (
        scope_to_tenant(
            Incident.objects.select_related("assigned_opd", "regency", "issue"), request
        )
        .prefetch_related("events", "mentions__platform")
        .filter(pk=incident_id)
        .first()
    )
```
